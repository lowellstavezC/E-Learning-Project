Table Role {

    password varchar
    role_id int [ref: > Role.id]
    created_at datetime
    updated_at datetime
}

    id int [primary key]
    user_id int [ref: > User.id]
    created_at datetime
    updated_at datetime
}

    id int [primary key]

    department varchar
    created_at datetime
    updated_at datetime
}

    id int [primary key]
    user_id int [ref: > User.id]
    course_of_study varchar
    created_at datetime
    updated_at datetime
}

Table CourseCategory {
    id int [primary key]
    name varchar
    description text
    created_at datetime
    updated_at datetime
}

Table Course {
    id int [primary key]
    title varchar
    description text
    category_id int [ref: > CourseCategory.id]
    teacher_id int [ref: > Teacher.id]

    updated_at datetime
}

    id int [primary key]

    content text
